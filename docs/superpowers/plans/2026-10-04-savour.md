# Savour Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `Savour`, the third portfolio showcase site at `/showcase/savour`: a Pune food guide where visitors browse real restaurants and their dishes, save individual dishes, and organise them into private per-visitor lists backed by Supabase Postgres with RLS.

**Architecture:** Catalogue reads happen in Server Components against Supabase (public `select`-only policies, `revalidate = 3600`), with a bundled 3-venue fallback so a misconfigured deployment still renders. Wishlist writes happen in Client Components using Supabase **Anonymous sign-ins**, so `auth.uid()` is a server-issued per-visitor id and Row Level Security makes each visitor's rows invisible to every other visitor. Non-component modules live inside `components/sites/savour/` beside the components, matching how Market Pulse keeps `types.ts`, `format.ts`, `indicators.ts` and `useStocks.ts` inside `components/sites/stock-screener/`. No new top-level `lib/` directory.

**Tech Stack:** Next.js 15.5 App Router, React 19, TypeScript 5.7 `strict`, Tailwind CSS 3.4 (scoped `.savour` CSS-variable theme in `app/globals.css`), `lucide-react`, `@supabase/supabase-js`, Vitest + `tsx` (dev-only).

**Spec:** `docs/superpowers/specs/2026-10-04-savour-design.md` — read it alongside this plan. The spec's appendix holds the exact DDL **already applied** to the live Supabase project; Task 1 copies it into the repo.

## Global constraints

- **No secret in the client bundle.** `NEXT_PUBLIC_SAVOUR_SUPABASE_URL` and `NEXT_PUBLIC_SAVOUR_SUPABASE_PUBLISHABLE_KEY` are the only Supabase vars this app reads, and both are public-safe. The Supabase **service-role / `secret` key must never appear in `.env.local`, in any file under `components/`, `app/`, `scripts/`, or in generated output**. Seed SQL is applied by Jainil in the Dashboard, not by code.
- **`NVIDIA_API_KEY` stays server-side** (no `NEXT_PUBLIC_` prefix, read only from `app/api/**`). Savour does not call NVIDIA NIM at all.
- **Real keys go only in `.env.local`** (gitignored by `.gitignore` `.env.*`). `.env.example` holds placeholders only.
- **Do not stage unrelated work.** `git status` shows ~25 modified/untracked Hair Adore and stock-screener files. Every commit here stages only the paths it names, by explicit path. Never `git add -A` or `git add .`.
- **`git push` and Vercel deploys require asking Jainil first.** Task 15 is the only task that pushes.
- **Test scope:** Vitest covers pure logic only — env accessor, Overpass transform, catalogue validation, seed SQL, dish join, filter/sort pipeline, wishlist reducer. Components are verified in the browser. **No React Testing Library, no jsdom.**
- **UI rule from Jainil: no tabs.** Every list is filterable and fully visible on one page. Cuisine/price/veg controls filter an already-rendered grid; they are never panel-swapping tabs. Venue hours render as per-day rows, not day tabs.
- **Copy rules:** prices in rupees with `₹`, no decimals for whole numbers. Each rating appears with its venue's `verified_at` date nearby. `positive_reviews` is labelled "estimate". The home footer carries the OpenStreetMap attribution and states menus are curated.
- **Style:** match the repo — single quotes, semicolons, 2-space indent.
- **Node ≥ 20** (`package.json` engines; session has v24.18.0). Run all commands from the repo root.

```ts
const PLACEHOLDER = /YOUR-PROJECT-REF|\bxxxxx\b/i;

export interface SavourEnv {
  url: string;
  key: string;
}

export function readSavourEnv(): SavourEnv | null {
  const url = (process.env.NEXT_PUBLIC_SAVOUR_SUPABASE_URL ?? '').trim();
  const key = (process.env.NEXT_PUBLIC_SAVOUR_SUPABASE_PUBLISHABLE_KEY ?? '').trim();
  if (!url || !key) return null;
  if (PLACEHOLDER.test(url)) return null;
  if (!/^https:\/\//.test(url)) return null;
  return { url, key };
}

export function isSavourConfigured(): boolean {
  return readSavourEnv() !== null;
}
```

```ts
export const FOOD_AMENITIES = new Set(['restaurant', 'cafe', 'fast_food', 'food_court', 'bar', 'pub']);

export const CUISINE_RENAMES: Record<string, string> = {
  coffee_shop: 'coffee',
  regional: 'maharashtrian',
  all_in_caffe: 'cafe',
  ice_lemon_tea: 'tea',
};

const BAND_BY_AMENITY: Record<string, PriceBand> = {
  restaurant: '₹₹₹',
  cafe: '₹₹',
  fast_food: '₹',
  food_court: '₹',
  bar: '₹₹₹₹',
  pub: '₹₹₹',
};

const DAYS = ['Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa', 'Su'];

export function slugify(name: string, area?: string | null): string {
  return `${name}-${area ?? ''}`
    .toLowerCase()
    .normalize('NFKD')
    .replace(/['’`]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

export function parseCuisines(raw: string | undefined): string[] {
  if (!raw) return [];
  const out: string[] = [];
  for (const piece of raw.split(/[;,]/)) {
    const t = piece.trim().toLowerCase().replace(/\s+/g, '_');
    if (!t) continue;
    const renamed = CUISINE_RENAMES[t] ?? t;
    if (!out.includes(renamed)) out.push(renamed);
  }
  return out;
}

function expandDays(spec: string): string[] {
  const s = spec.trim();
  const norm = (x: string) => x[0].toUpperCase() + x.slice(1).toLowerCase();
  if (/^(mo-su|daily|week)$/i.test(s)) return [...DAYS];
  const range = s.match(/^([A-Za-z]{2})-([A-Za-z]{2})$/);
  if (range) {
    const a = DAYS.indexOf(norm(range[1]));
    const b = DAYS.indexOf(norm(range[2]));
    if (a !== -1 && b !== -1 && b >= a) return DAYS.slice(a, b + 1);
    return [];
  }
  return /^[A-Za-z]{2}$/.test(s) && DAYS.includes(norm(s)) ? [norm(s)] : [];
}

const TIME_BLOCK = /^\d{1,2}:\d{2}(?:-\d{1,2}:\d{2})?(?:,\d{1,2}:\d{2}-\d{1,2}:\d{2})*$/;

export function parseHours(raw: string | undefined): Record<string, string[]> | null {
  if (!raw) return null;
  const out: Record<string, string[]> = {};
  let matched = false;
  for (const block of raw.split(';')) {
    const b = block.trim();
    if (!b || b.toUpperCase() === 'CLOSED') continue;
    const parts = b.split(/\s+/);
    if (parts.length < 2) continue;
    const times = parts[parts.length - 1];
    if (!TIME_BLOCK.test(times)) continue;
    const days = parts.slice(0, -1).flatMap(expandDays);
    if (days.length === 0) continue;
    matched = true;
    for (const d of days) {
      const list = out[d] ?? (out[d] = []);
      for (const t of times.split(',')) if (!list.includes(t)) list.push(t);
    }
  }
  if (!matched) return null;
  const ordered: Record<string, string[]> = {};
  for (const d of DAYS) if (out[d]) ordered[d] = out[d];
  return ordered;
}

export function mapOverpassToPlace(el: OverpassElement): PlaceRow | null {
  const t = el.tags ?? {};
  const name = (t['name'] ?? t['int_name'] ?? '').trim();
  const kind = t['amenity'];
  if (!name || !kind || !FOOD_AMENITIES.has(kind)) return null;

  const lat = num(el.lat ?? el.center?.lat);
  const lng = num(el.lon ?? el.center?.lon);
  if (lat === null || lng === null) return null;

  const area = t['addr:suburb'] ?? t['addr:street'] ?? null;
  const slug = slugify(name, area);
  if (!slug) return null;

  return {
    id: `${el.type[0]}${el.id}`,
    slug,
    name,
    city: t['addr:city'] ?? 'Pune',
    state: t['addr:state'] ?? 'Maharashtra',
    area,
    address: t['addr:full'] ?? null,
    lat,
    lng,
    cuisines: parseCuisines(t['cuisine'] ?? t['food'] ?? t['cuisine:description']),
    price_band: BAND_BY_AMENITY[kind],
    cost_for_two: null,
    rating: null,
    num_reviews: 0,
    positive_reviews: 0,
    phone: t['phone'] ?? t['contact:phone'] ?? null,
    website: t['website'] ?? t['contact:website'] ?? null,
    instagram: t['contact:instagram'] ?? null,
    hours: parseHours(t['opening_hours']),
    image_url: null,
    data_source: 'geo-api',
    verified_at: null,
  };
}
```

```ts
import { mkdir, writeFile } from 'node:fs/promises';
import path from 'node:path';
import { QUERY, mapOverpassToPlace, type OverpassElement } from '../components/sites/savour/osm';
import type { PlaceRow } from '../components/sites/savour/types';

const ENDPOINT = 'https://overpass-api.de/api/interpreter';
const CITY = process.argv[2] ?? 'Pune';
const OUT_DIR = path.join(import.meta.dirname, 'out');

/** Prefer venues with identity: hours, phone and website beat a bare name. */
function richness(p: PlaceRow): number {
  return (
    (p.hours ? 4 : 0) + (p.phone ? 2 : 0) + (p.website ? 2 : 0) + (p.cuisines.length ? 2 : 0) + (p.address ? 1 : 0)
  );
}

async function main() {
  const res = await fetch(ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded',
      // Without this header Overpass answers HTTP 406. Measured 2026-10-04.
      Accept: 'application/json',
      'User-Agent': 'savour-portfolio-demo/1.0',
    },
    body: new URLSearchParams({ data: QUERY(CITY) }).toString(),
  });
  if (!res.ok) throw new Error(`Overpass ${res.status}: ${(await res.text()).slice(0, 300)}`);
  const json = (await res.json()) as { elements: OverpassElement[] };

  const seen = new Map<string, PlaceRow>();
  for (const el of json.elements) {
    const p = mapOverpassToPlace(el);
    if (p && !seen.has(p.slug)) seen.set(p.slug, p);
  }

  const rows = [...seen.values()].sort(
    (a, b) => richness(b) - richness(a) || a.name.localeCompare(b.name),
  );
  await mkdir(OUT_DIR, { recursive: true });
  const file = path.join(OUT_DIR, `${CITY.toLowerCase()}-places.json`);
  await writeFile(file, `${JSON.stringify(rows, null, 2)}\n`, 'utf8');
  console.log(`${rows.length} venues -> ${file}`);
  console.log(`with hours: ${rows.filter((r) => r.hours).length}, with cuisine: ${rows.filter((r) => r.cuisines.length).length}`);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```ts
import type { Course, MenuItemRow, PlaceRow, PriceBand, ReviewRow } from './types';

export type PlaceSeed = Omit<PlaceRow, 'id'>;
export type DishSeed = Omit<MenuItemRow, 'id' | 'place_id'>;
export type ReviewSeed = Omit<ReviewRow, 'id' | 'place_id'>;

export interface VenueSeed {
  place: PlaceSeed;
  dishes: DishSeed[];
  reviews: ReviewSeed[];
}

/** Inclusive low, exclusive high per band: sanity-checks that ₹₹₹₹ means fine dining. */
export const BAND_PRICE_CEILING: Record<PriceBand, [number, number]> = {
  '₹': [0, 400],
  '₹₹': [100, 800],
  '₹₹₹': [300, 1800],
  '₹₹₹₹': [600, 6000],
};

const COURSES: Course[] = ['aperitif', 'starter', 'main', 'breads', 'dessert', 'beverage'];
const SLUG_RE = /^[a-z0-9]+(-[a-z0-9]+)*$/;
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

export function validateCatalog(catalog: VenueSeed[]): string[] {
  const errs: string[] = [];
  if (catalog.length !== 12) errs.push(`expected 12 venues, got ${catalog.length}`);
  const slugs = new Set<string>();
  for (const v of catalog) {
    const p = v.place;
    const label = p.slug || p.name || '<unnamed>';
    if (!SLUG_RE.test(p.slug)) errs.push(`${label}: slug "${p.slug}" is not kebab-case`);
    if (slugs.has(p.slug)) errs.push(`${label}: duplicate slug`);
    slugs.add(p.slug);
    if (!p.name.trim()) errs.push(`${label}: name is required`);
    if (p.city !== 'Pune') errs.push(`${label}: city must be Pune, got "${p.city}"`);
    if (p.state !== 'Maharashtra') errs.push(`${label}: state must be Maharashtra`);
    if (p.rating !== null && (p.rating < 0 || p.rating > 5)) errs.push(`${label}: rating out of 0-5`);
    if (p.positive_reviews > p.num_reviews) errs.push(`${label}: positive_reviews exceeds num_reviews`);
    if (p.verified_at !== null && !DATE_RE.test(p.verified_at)) errs.push(`${label}: verified_at must be YYYY-MM-DD`);
    if (p.lat === null || p.lng === null) errs.push(`${label}: lat/lng required`);
    else if (p.lat < 18.3 || p.lat > 18.7 || p.lng < 73.6 || p.lng > 74.1)
      errs.push(`${label}: coords outside Pune bounding box`);
    if (!(p.price_band in BAND_PRICE_CEILING)) errs.push(`${label}: unknown price_band`);

    if (v.dishes.length < 8) errs.push(`${label}: needs at least 8 dishes`);
    if (!v.dishes.some((d) => d.is_signature)) errs.push(`${label}: needs at least 1 signature dish`);
    if (v.reviews.length < 3) errs.push(`${label}: needs at least 3 reviews`);

    const names = new Set<string>();
    const [low, high] = BAND_PRICE_CEILING[p.price_band] ?? [0, Infinity];
    for (const d of v.dishes) {
      if (names.has(d.name)) errs.push(`${label}: duplicate dish "${d.name}"`);
      names.add(d.name);
      if (d.name.length > 80) errs.push(`${label}: dish name over 80 chars`);
      if (!COURSES.includes(d.course)) errs.push(`${label}: bad course "${d.course}" on "${d.name}"`);
      if (!(d.price > 0)) errs.push(`${label}: dish "${d.name}" needs a positive price`);
      if (d.price >= high) errs.push(`${label}: dish "${d.name}" price ${d.price} out of band ${p.price_band}`);
      if (d.price < low) errs.push(`${label}: dish "${d.name}" price ${d.price} below band ${p.price_band}`);
    }
    for (const r of v.reviews) {
      if (!r.author.trim()) errs.push(`${label}: review author required`);
      if (r.rating < 1 || r.rating > 5) errs.push(`${label}: review rating out of 1-5`);
      if (r.body.trim().length < 20) errs.push(`${label}: review body too short`);
      if (r.visited_on !== null && !DATE_RE.test(r.visited_on))
        errs.push(`${label}: review visited_on must be YYYY-MM-DD`);
    }
  }
  return errs;
}
```

```ts
import path from 'node:path';
import { CATALOG, validateCatalog, type VenueSeed } from '../components/sites/savour/catalog';

/** Stable uuid per input, so re-running the seed updates rows instead of duplicating them. */
export function gen(seed: string): string {
  const h = createHash('sha1')
    .update(`savour:${seed}`)
    .digest('hex');
  return `${h.slice(0, 8)}-${h.slice(8, 12)}-4${h.slice(13, 16)}-8${h.slice(17, 20)}-${h.slice(21, 32)}`;
}

export function q(v: string | number | null | undefined): string {
  if (v === null || v === undefined) return 'null';
  if (typeof v === 'number') return String(v);
  return `'${v.replace(/'/g, "''")}'`;
}

function textArray(items: string[]): string {
  if (items.length === 0) return "'{}'::text[]";
  return `array[${items.map((i) => q(i)).join(', ')}]::text[]`;
}

const PLACE_COLS =
  'id, slug, name, city, state, area, address, lat, lng, cuisines, price_band, cost_for_two, rating, num_reviews, positive_reviews, phone, website, instagram, hours, image_url, data_source, verified_at';

export function buildSeedSql(catalog: VenueSeed[]): string {
  const errs = validateCatalog(catalog);
  if (errs.length) throw new Error(`catalogue invalid:\n${errs.join('\n')}`);

  const out: string[] = [
    '-- Savour seed. Generated by scripts/savour-seed.ts — do not edit by hand.',
    '-- Idempotent: re-running updates the same rows instead of duplicating them.',
    '',
  ];

  for (const v of catalog) {
    const pid = gen(v.place.slug);
    const p = v.place;
    out.push(
      `-- venue: ${p.name}`,
      `insert into public.places (${PLACE_COLS})`,
      'values',
      [
        `  (${q(pid)}, ${q(p.slug)}, ${q(p.name)}, ${q(p.city)}, ${q(p.state)},`,
        `   ${q(p.area)}, ${q(p.address)}, ${p.lat ?? 'null'}, ${p.lng ?? 'null'},`,
        `   ${textArray(p.cuisines)}, ${q(p.price_band)}, ${p.cost_for_two ?? 'null'},`,
        `   ${p.rating ?? 'null'}, ${p.num_reviews}, ${p.positive_reviews},`,
        `   ${q(p.phone)}, ${q(p.website)}, ${q(p.instagram)},`,
        `   ${p.hours === null ? 'null' : `${q(JSON.stringify(p.hours))}::jsonb`}, ${q(p.image_url)},`,
        `   ${q(p.data_source)}, ${q(p.verified_at)})`,
      ].join('\n'),
      'on conflict (slug) do update set',
      '  name = excluded.name, area = excluded.area, address = excluded.address,',
      '  lat = excluded.lat, lng = excluded.lng, cuisines = excluded.cuisines,',
      '  price_band = excluded.price_band, cost_for_two = excluded.cost_for_two,',
      '  rating = excluded.rating, num_reviews = excluded.num_reviews,',
      '  positive_reviews = excluded.positive_reviews, phone = excluded.phone,',
      '  website = excluded.website, instagram = excluded.instagram, hours = excluded.hours,',
      '  image_url = excluded.image_url, data_source = excluded.data_source,',
      '  verified_at = excluded.verified_at;',
      '',
      'insert into public.menu_items (id, place_id, name, course, course_rank, description, price, is_veg, is_signature)',
      'values',
      v.dishes
        .map(
          (d) =>
            `  (${q(gen(`${p.slug}#dish#${d.name}`))}, ${q(pid)}, ${q(d.name)}, ${q(d.course)}, ${d.course_rank}, ` +
            `${q(d.description)}, ${d.price}, ${d.is_veg === null ? 'null' : d.is_veg ? 'true' : 'false'}, ${d.is_signature ? 'true' : 'false'});`,
        )
        .join('\n'),
      'on conflict (place_id, name) do update set',
      '  course = excluded.course, course_rank = excluded.course_rank,',
      '  description = excluded.description, price = excluded.price,',
      '  is_veg = excluded.is_veg, is_signature = excluded.is_signature;',
      '',
      'insert into public.reviews (id, place_id, author, rating, body, visited_on, source)',
      'values',
      v.reviews
        .map(
          (r) =>
            `  (${q(gen(`${p.slug}#review#${r.author}#${r.body.slice(0, 24)}`))}, ${q(pid)}, ${q(r.author)}, ` +
            `${r.rating}, ${q(r.body)}, ${q(r.visited_on)}, ${q(r.source)});`,
        )
        .join('\n'),
      'on conflict (id) do nothing;',
      '',
    );
  }
  return out.join('\n');
}

if (process.argv[1] && path.basename(process.argv[1]) === 'savour-seed.ts') {
  await writeFile(path.join(import.meta.dirname, '..', 'supabase', 'seed.sql'), buildSeedSql(CATALOG), 'utf8');
  console.log('wrote supabase/seed.sql');
}
```

```ts
import type { DishCard, MenuItemRow, PlaceRow } from './types';

export function buildDishCards(places: PlaceRow[], items: MenuItemRow[]): DishCard[] {
  const byId = new Map(places.map((p) => [p.id, p]));
  const cards: DishCard[] = [];
  for (const item of items) {
    const place = byId.get(item.place_id);
    if (place) cards.push({ item, place });
  }
  return cards;
}
```

```ts
import { CATALOG, type VenueSeed } from './catalog';
import type { Catalogue, MenuItemRow, PlaceRow, ReviewRow } from './types';

/**
 * First three catalogue venues, inlined as rows, for when Supabase is
 * unconfigured or unreachable. `degraded: true` makes the UI say so.
 * Ids come from the same gen() the seed uses, so a saved dish id stays valid
 * if the same browser moves between degraded and live modes.
 */
function seedToRows(v: VenueSeed): { place: PlaceRow; items: MenuItemRow[]; reviews: ReviewRow[] } {
  const id = gen(v.place.slug);
  return {
    place: { ...v.place, id },
    items: v.dishes.map((d) => ({ ...d, id: gen(`${v.place.slug}#dish#${d.name}`), place_id: id })),
    reviews: v.reviews.map((r) => ({
      ...r,
      id: gen(`${v.place.slug}#review#${r.author}#${r.body.slice(0, 24)}`),
      place_id: id,
    })),
  };
}

const three = CATALOG.slice(0, 3).map(seedToRows);

  places: three.map((t) => t.place),
  items: three.flatMap((t) => t.items),
  reviews: three.flatMap((t) => t.reviews),
```

```ts
import 'server-only';
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { readSavourEnv } from './env';
import { buildDishCards } from './dishes';
import { FALLBACK } from './fallback-data';
import {
  bool, int, num,
  type Catalogue, type MenuItemRow, type PlacePageData, type PlaceRow, type ReviewRow,
} from './types';

const PLACE_COLUMNS =
  'id, slug, name, city, state, area, address, lat, lng, cuisines, price_band, cost_for_two, ' +
  'rating, num_reviews, positive_reviews, phone, website, instagram, hours, image_url, data_source, verified_at';
const ITEM_COLUMNS =
  'id, place_id, name, course, course_rank, description, price, is_veg, is_signature, image_url';
const REVIEW_COLUMNS = 'id, place_id, author, rating, body, visited_on, source';

function client(): SupabaseClient | null {
  const env = readSavourEnv();
  if (!env) return null;
  return createClient(env.url, env.key, { auth: { persistSession: false, autoRefreshToken: false } });
}

function shapePlace(r: Record<string, unknown>): PlaceRow {
  return {
    ...(r as unknown as PlaceRow),
    lat: num(r.lat as string),
    lng: num(r.lng as string),
    cost_for_two: num(r.cost_for_two as string),
    rating: num(r.rating as string),
    num_reviews: int(r.num_reviews as string),
    positive_reviews: int(r.positive_reviews as string),
    cuisines: Array.isArray(r.cuisines) ? (r.cuisines as string[]) : [],
    hours: (r.hours as Record<string, string[]> | null) ?? null,
  };
}

function shapeItem(r: Record<string, unknown>): MenuItemRow {
  return {
    ...(r as unknown as MenuItemRow),
    price: num(r.price as string) ?? 0,
    is_veg: r.is_veg === null || r.is_veg === undefined ? null : bool(r.is_veg as string),
  };
}

function shapeReview(r: Record<string, unknown>): ReviewRow {
  return { ...(r as unknown as ReviewRow), rating: int(r.rating as string) };
}

type Result<T> = { data: T[] | null; error: unknown };

async function rows<T>(run: () => Promise<Result<Record<string, unknown>>>, shape: (r: Record<string, unknown>) => T): Promise<T[] | null> {
  try {
    const { data, error } = await run();
    if (error || !data) return null;
    return data.map(shape);
  } catch {
    return null;
  }
}

export async function getCatalogue(): Promise<Catalogue> {
  const sb = client();
  if (!sb) return FALLBACK;
  const [places, items, reviews] = await Promise.all([
    rows(() => sb.from('places').select(PLACE_COLUMNS).eq('state', 'Maharashtra').order('rating', { ascending: false }).limit(60) as never, shapePlace),
    rows(() => sb.from('menu_items').select(ITEM_COLUMNS).order('course_rank').limit(1200) as never, shapeItem),
    rows(() => sb.from('reviews').select(REVIEW_COLUMNS).order('rating', { ascending: false }).limit(240) as never, shapeReview),
  ]);
  if (!places || places.length === 0) return FALLBACK;
  return { places, items: items ?? [], reviews: reviews ?? [], degraded: false };
}

/**
 * One code path serves live and degraded mode: read the whole catalogue
 * (already the home page's query) and filter. The extra round trip is fine
 * because every caller is an ISR-cached page with revalidate = 3600.
 */
export async function getPlaceBySlug(slug: string): Promise<PlacePageData | null> {
  const cat = await getCatalogue();
  const place = cat.places.find((p) => p.slug === slug);
  if (!place) return null;
  return {
    place,
    items: cat.items.filter((i) => i.place_id === place.id),
    reviews: cat.reviews.filter((r) => r.place_id === place.id),
    degraded: cat.degraded,
  };
}

export { buildDishCards };
```

```ts
import type { DishCard, Filters, PriceBand, SortKey } from './types';

const BANDS: PriceBand[] = ['₹', '₹₹', '₹₹₹', '₹₹₹₹'];

export function bandRank(b: PriceBand): number {
  return BANDS.indexOf(b) + 1;
}

export function availableCuisines(cards: DishCard[]): string[] {
  const set = new Set<string>();
  for (const c of cards) for (const cu of c.place.cuisines) set.add(cu);
  return [...set].sort();
}

export function availableCities(cards: DishCard[]): string[] {
  const set = new Set<string>();
  for (const c of cards) set.add(c.place.city);
  return [...set].sort();
}

const byName = (a: DishCard, b: DishCard) => a.item.name.localeCompare(b.item.name);
const byRating = (a: DishCard, b: DishCard) => (b.place.rating ?? -1) - (a.place.rating ?? -1) || byName(a, b);

const SORTERS: Record<SortKey, (a: DishCard, b: DishCard, saved: Set<string>) => number> = {
  rating: (a, b) => byRating(a, b),
  name: (a, b) => byName(a, b),
  'price-asc': (a, b) => a.item.price - b.item.price || byName(a, b),
  'price-desc': (a, b) => b.item.price - a.item.price || byName(a, b),
  saved: (a, b, saved) =>
    Number(saved.has(b.item.id)) - Number(saved.has(a.item.id)) || byRating(a, b),
};

export function applyFilters(cards: DishCard[], f: Filters, saved: Set<string>): DishCard[] {
  const q = f.q.trim().toLowerCase();
  return cards
    .filter((c) => {
      if (f.city && c.place.city !== f.city) return false;
      if (f.cuisines.length && !f.cuisines.some((cu) => c.place.cuisines.includes(cu))) return false;
      if (f.bands.length && !f.bands.includes(c.place.price_band)) return false;
      if (f.vegOnly && c.item.is_veg !== true) return false;
      if (q) {
        const hay = `${c.item.name} ${c.item.description ?? ''} ${c.place.name}`.toLowerCase();
        if (!hay.includes(q)) return false;
      }
      return true;
    })
    .sort((a, b) => SORTERS[f.sort](a, b, saved));
}
```

```ts
import type { WishItem, WishlistListRow } from './types';

export interface WishlistState {
  lists: WishlistListRow[];
  items: WishItem[];
  /** tempId -> in-flight kind, for optimistic rows and their rollback. */
  pending: Record<string, 'saving' | 'deleting'>;
  defaultListId: string | null;
  error: string | null;
}

export const EMPTY_WISHLIST: WishlistState = {
  lists: [],
  items: [],
  pending: {},
  defaultListId: null,
  error: null,
};

export type WishlistAction =
  | { type: 'hydrate'; lists: WishlistListRow[]; items: WishItem[] }
  | { type: 'save-pending'; tempId: string; listId: string; menuItemId: string }
  | { type: 'save-committed'; tempId: string; item: WishItem }
  | { type: 'save-failed'; tempId: string; message?: string }
  | { type: 'remove-committed'; id: string }
  | { type: 'move-committed'; id: string; listId: string }
  | { type: 'set-note'; id: string; note: string | null }
  | { type: 'toggle-tried'; id: string }
  | { type: 'add-list'; list: WishlistListRow }
  | { type: 'rename-list'; id: string; name: string }
  | { type: 'set-list-note'; id: string; note: string | null }
  | { type: 'delete-list'; id: string }
  | { type: 'reset-failed' };

const withoutPending = (pending: Record<string, 'saving' | 'deleting'>, id: string) => {
  const rest = { ...pending };
  delete rest[id];
  return rest;
};

/**
 * Rollback target for each optimistic save, keyed by the state it produced.
 * Held outside the state so a snapshot never leaks into rendering.
 */
const snapshots = new WeakMap<WishlistState, WishItem[]>();

export function wishlistReducer(state: WishlistState, action: WishlistAction): WishlistState {
  switch (action.type) {
    case 'hydrate': {
      const def = action.lists.find((l) => l.is_default) ?? action.lists[0];
      return { lists: action.lists, items: action.items, pending: {}, defaultListId: def?.id ?? null, error: null };
    }
    case 'save-pending': {
      if (state.items.some((i) => i.list_id === action.listId && i.menu_item_id === action.menuItemId)) return state;
      const optimistic: WishItem = {
        id: action.tempId,
        user_id: '',
        list_id: action.listId,
        menu_item_id: action.menuItemId,
        note: null,
        tried: false,
        dish: null,
      };
      const next: WishlistState = {
        ...state,
        items: [...state.items, optimistic],
        pending: { ...state.pending, [action.tempId]: 'saving' },
        error: null,
      };
      snapshots.set(next, state.items);
      return next;
    }
    case 'save-committed':
      return {
        ...state,
        items: state.items.map((i) => (i.id === action.tempId ? action.item : i)),
        pending: withoutPending(state.pending, action.tempId),
      };
    case 'save-failed': {
      const before = snapshots.get(state) ?? state.items.filter((i) => i.id !== action.tempId);
      return {
        ...state,
        items: before.filter((i) => i.id !== action.tempId),
        pending: withoutPending(state.pending, action.tempId),
        error: action.message ?? 'Could not save. Retry from the wishlist.',
      };
    }
    case 'remove-committed':
      return { ...state, items: state.items.filter((i) => i.id !== action.id) };
    case 'move-committed':
      return { ...state, items: state.items.map((i) => (i.id === action.id ? { ...i, list_id: action.listId } : i)) };
    case 'set-note':
      return { ...state, items: state.items.map((i) => (i.id === action.id ? { ...i, note: action.note } : i)) };
    case 'toggle-tried':
      return { ...state, items: state.items.map((i) => (i.id === action.id ? { ...i, tried: !i.tried } : i)) };
    case 'add-list':
      return { ...state, lists: [...state.lists, action.list], defaultListId: state.defaultListId ?? action.list.id };
    case 'rename-list':
      return { ...state, lists: state.lists.map((l) => (l.id === action.id ? { ...l, name: action.name } : l)) };
    case 'set-list-note':
      return { ...state, lists: state.lists.map((l) => (l.id === action.id ? { ...l, note: action.note } : l)) };
    case 'delete-list': {
      if (state.lists.length <= 1) return { ...state, error: 'Keep at least one list.' };
      const lists = state.lists.filter((l) => l.id !== action.id);
      const nextDefault = lists.find((l) => l.is_default) ?? lists[0];
      return {
        ...state,
        lists: lists.map((l) => ({ ...l, is_default: l.id === nextDefault?.id })),
        items: state.items.filter((i) => i.list_id !== action.id),
        defaultListId: nextDefault?.id ?? null,
        error: null,
      };
    }
    case 'reset-failed':
      return { ...state, error: null };
    default:
      return state;
  }
}

export function savedItemIds(state: WishlistState): Set<string> {
  return new Set(state.items.map((i) => i.menu_item_id));
}
```

```tsx
'use client';
import {
  createContext, useCallback, useContext, useEffect, useMemo, useReducer, useRef, useState,
  type ReactNode,
} from 'react';
import { getSavourClient } from './browser';
import { isSavourConfigured } from './env';
import {
  EMPTY_WISHLIST, savedItemIds, wishlistReducer,
} from './wishlist-reducer';
import type { DishRef, MenuItemRow, PlaceRow, WishItem, WishlistListRow } from './types';

/** PostgREST many-to-one embeds: menu_items -> places is a single object. */
const ITEM_SELECT =
  'id, user_id, list_id, menu_item_id, note, tried, ' +
  'menu_items(id, name, price, course, place:places(slug, name))';

interface RawItem extends Omit<WishItem, 'dish'> {
  menu_items: (Pick<MenuItemRow, 'id' | 'name' | 'price' | 'course'> & {
    place: Pick<PlaceRow, 'slug' | 'name'>;
  }) | null;
}

function toWishItem(r: RawItem): WishItem {
  const dish: DishRef | null = r.menu_items
    ? {
        id: r.menu_items.id,
        name: r.menu_items.name,
        price: Number(r.menu_items.price),
        course: r.menu_items.course,
        place_slug: r.menu_items.place?.slug ?? '',
        place_name: r.menu_items.place?.name ?? '',
      }
    : null;
  return {
    id: r.id, user_id: r.user_id, list_id: r.list_id, menu_item_id: r.menu_item_id,
    note: r.note, tried: r.tried, dish,
  };
}

interface WishlistCtx {
  enabled: boolean;
  loading: boolean;
  lists: WishlistListRow[];
  items: WishItem[];
  defaultListId: string | null;
  error: string | null;
  saved: Set<string>;
  save: (menuItemId: string, listId?: string) => Promise<void>;
  unsave: (wishId: string) => Promise<void>;
  move: (wishId: string, listId: string) => Promise<void>;
  setNote: (wishId: string, note: string | null) => Promise<void>;
  toggleTried: (wishId: string) => Promise<void>;
  createList: (name: string) => Promise<string | null>;
  renameList: (id: string, name: string) => Promise<void>;
  setListNote: (id: string, note: string | null) => Promise<void>;
  deleteList: (id: string) => Promise<void>;
  retry: () => Promise<void>;
}

const Ctx = createContext<WishlistCtx | null>(null);

export function useWishlist(): WishlistCtx {
  const c = useContext(Ctx);
  if (!c) throw new Error('useWishlist must be used inside <WishlistProvider>');
  return c;
}

export function WishlistProvider({ children }: { children: ReactNode }) {
  const enabled = isSavourConfigured();
  const [state, dispatch] = useReducer(wishlistReducer, EMPTY_WISHLIST);
  const [loading, setLoading] = useState(enabled);
  const seq = useRef(0);
  const tempId = () => `T${++seq.current}`;

  /** The anonymous uid is read from the session; never from the caller. */
  const uid = async (): Promise<string | null> => {
    const sb = getSavourClient();
    if (!sb) return null;
    const { data } = await sb.auth.getSession();
    if (data.session) return data.session.user.id;
    const { data: signed, error } = await sb.auth.signInAnonymously();
    if (error) throw error;
    return signed.user?.id ?? null;
  };

  const load = useCallback(async () => {
    const sb = getSavourClient();
    if (!sb) return;
    setLoading(true);
    try {
      const id = await uid();
      if (!id) throw new Error('Anonymous sign-in is disabled on this Supabase project.');
      const [L, I] = await Promise.all([
        sb.from('wishlist_lists').select('*').order('position'),
        sb.from('wishlist_items').select(ITEM_SELECT).order('created_at'),
      ]);
      if (L.error || I.error) throw L.error ?? I.error;
      let lists = (L.data ?? []) as WishlistListRow[];
      const items = ((I.data ?? []) as unknown as RawItem[]).map(toWishItem);
      if (lists.length === 0) {
        const { data: created } = await sb
          .from('wishlist_lists')
          .insert({ user_id: id, name: 'Want to try', is_default: true, position: 0 })
          .select('*')
          .single();
        if (created) lists = [created as WishlistListRow];
      }
      dispatch({ type: 'hydrate', lists, items });
    } catch (e) {
      dispatch({ type: 'hydrate', lists: [], items: [] });
      dispatch({
        type: 'save-failed',
        tempId: 'boot',
        message: e instanceof Error ? e.message : 'Wishlist unavailable.',
      });
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    if (enabled) void load();
  }, [enabled, load]);

  const save = useCallback(
    async (menuItemId: string, listId?: string) => {
      const sb = getSavourClient();
      const target = listId ?? state.defaultListId;
      if (!sb || !target) return;
      const id = await uid().catch(() => null);
      if (!id) return;
      const temp = tempId();
      dispatch({ type: 'save-pending', tempId: temp, listId: target, menuItemId });
      const { data, error } = await sb
        .from('wishlist_items')
        .insert({ user_id: id, list_id: target, menu_item_id: menuItemId })
        .select(ITEM_SELECT)
        .single();
      if (error || !data) dispatch({ type: 'save-failed', tempId: temp, message: error?.message });
      else dispatch({ type: 'save-committed', tempId: temp, item: toWishItem(data as unknown as RawItem) });
    },
    [state.defaultListId],
  );

  const unsave = useCallback(async (wishId: string) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_items').delete().eq('id', wishId);
    if (error) dispatch({ type: 'save-failed', tempId: wishId, message: error.message });
    else dispatch({ type: 'remove-committed', id: wishId });
  }, []);

  const move = useCallback(async (wishId: string, listId: string) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_items').update({ list_id: listId }).eq('id', wishId);
    if (!error) dispatch({ type: 'move-committed', id: wishId, listId });
  }, []);

  const setNote = useCallback(async (wishId: string, note: string | null) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_items').update({ note }).eq('id', wishId);
    if (!error) dispatch({ type: 'set-note', id: wishId, note });
  }, []);

  const toggleTried = useCallback(
    async (wishId: string) => {
      const sb = getSavourClient();
      const cur = state.items.find((i) => i.id === wishId);
      if (!sb || !cur) return;
      dispatch({ type: 'toggle-tried', id: wishId });
      const { error } = await sb.from('wishlist_items').update({ tried: !cur.tried }).eq('id', wishId);
      if (error) dispatch({ type: 'toggle-tried', id: wishId });
    },
    [state.items],
  );

  const createList = useCallback(
    async (name: string) => {
      const sb = getSavourClient();
      if (!sb) return null;
      const id = await uid().catch(() => null);
      if (!id) return null;
      const { data, error } = await sb
        .from('wishlist_lists')
        .insert({ user_id: id, name, position: state.lists.length })
        .select('*')
        .single();
      if (error || !data) return null;
      dispatch({ type: 'add-list', list: data as WishlistListRow });
      return (data as WishlistListRow).id;
    },
    [state.lists.length],
  );

  const renameList = useCallback(async (id: string, name: string) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_lists').update({ name }).eq('id', id);
    if (!error) dispatch({ type: 'rename-list', id, name });
  }, []);

  const setListNote = useCallback(async (id: string, note: string | null) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_lists').update({ note }).eq('id', id);
    if (!error) dispatch({ type: 'set-list-note', id, note });
  }, []);

  const deleteList = useCallback(async (id: string) => {
    const sb = getSavourClient();
    if (!sb) return;
    const { error } = await sb.from('wishlist_lists').delete().eq('id', id);
    if (!error) dispatch({ type: 'delete-list', id });
  }, []);

  const value = useMemo<WishlistCtx>(
    () => ({
      enabled,
      loading,
      lists: state.lists,
      items: state.items,
      defaultListId: state.defaultListId,
      error: state.error,
      saved: savedItemIds(state),
      save, unsave, move, setNote, toggleTried, createList, renameList, setListNote, deleteList,

  );

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
```

```tsx
export function SaveButton({ menuItemId, dishName, onSaved }: Props) {
  const w = useWishlist();
  const wish = w.items.find((i) => i.menu_item_id === menuItemId && !i.id.startsWith('T'));

      title={w.enabled ? label : 'Wishlist unavailable in this deployment'}

        if (saved && wish) void w.unsave(wish.id);
        else {
```
